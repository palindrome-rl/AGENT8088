# Agent8088 profile for Terminal-Bench 2.1 under Harbor.
# Loaded via AGENT8088_CONFIG inside the task container, together with
# AGENT8088_DISPOSABLE_CONTAINER=1. The Harbor adapter rewrites base_url and
# model from its --ak options at install time.

# --- Model: Qwen3.8-27B on vLLM (--max-model-len 32768) ---
default_provider=vllm
provider.vllm.base_url=http://host.docker.internal:8000/v1
provider.vllm.model=qwen3.8-27b
provider.vllm.api_mode=openai
provider.vllm.api_key_env=OPENAI_API_KEY
provider.vllm.native_tools=True
# Must equal vLLM's --max-model-len. Raise both together.
provider.vllm.context_window=32768
provider.vllm.max_completion_tokens=8192
# Sampling pinned to the vLLM server's own defaults on every request: a value
# agent8088 sends always overrides the server's, so each one is sent
# explicitly. Thinking on (server runs --reasoning-parser qwen3), top_p/top_k
# from the model's generation_config.json, penalties at the protocol default 0.
provider.vllm.extra_body={"chat_template_kwargs": {"enable_thinking": true}, "top_p": 0.95, "top_k": 20, "presence_penalty": 0.0, "frequency_penalty": 0.0}
# Thinking level: Terminus-2 ran with reasoning_effort=medium (Pi parity).
# Without this key the Qwen3.8 chat template defaults to xhigh.
provider.vllm.reasoning_effort=medium
model_name=qwen3.8-27b
model_base_url=http://host.docker.internal:8000/v1
context_window=32768
max_completion_tokens=8192
# A cut-off reply is retried at this limit, not double it: every reply is
# capped at 8192 tokens, as for Terminus-2 and Pi. (A doubled 16384-token
# retry at ~19 tok/s can take a whole 900 s task.)
length_retry_max_tokens=8192
# Per-call metadata (tokens, latency, finish reason, errors; never prompt or
# reply text) next to the trace, so Harbor collects it with agent/. Also makes
# vLLM report usage on streamed replies (the trace's token counts need it).
model_telemetry=1
model_telemetry_path=/logs/agent/agent8088-model-calls.jsonl
# The server's --override-generation-config temperature; agent8088's default
# 0.1 would otherwise override it.
temperature=0.6
presence_penalty=0
frequency_penalty=0
timeout_seconds=600

# --- Loop ---
max_turns=150
dynamic_turns_ceiling_multiplier=1
# Compact well before prompt + 8192 output reaches 32k.
compaction_threshold_pct=55
# Token estimate for Qwen: 2.8 characters per token, just below the 2.9-3.3
# measured on these conversations. agent8088's default of 4 undercounted by
# ~35% and could let a request pass vLLM's 32k limit before compaction fired;
# erring low keeps every request under 32k.
chars_per_token=2.8
# Output shown per tool result: 10,000 characters, the same window Terminus-2
# gets (its _limit_output_length, max_bytes=10000: first 5,000 + last 5,000).
# agent8088's default is 3,000. read_content continues in windows of the same size.
tool_result_max_chars=10000
read_content_window_chars=10000
# Agent8088's default tool-selection mode. With this profile's six-tool list
# (<= TOOL_SELECTION_TOP_K) no ranking occurs; the mode only adds the
# search_tools schema and deferred-tools routing language.
tool_selection=hybrid
# Structured per-turn JSON trace (messages + steps) next to the transcript.
# The adapter points AGENT8088_TRACE_DIR at the trial's agent log directory so
# Harbor collects it with the rest of agent/.
show_trace=1
# Memory ON, strictly the built-in native engine (SQLite, BM25 + vectors), never
# mem0. Each task starts in a fresh container (AGENT8088_HOME=/installed-agent/home),
# so the store begins empty. No embedding server in the container: facts are
# stored without vectors and found by BM25.
memory=1
memory_engine=native
plan_audit=0
reuse_known_verdicts=0
open_code_review_enabled=0
subagent_max_depth=0

# --- Tools: trimmed list, shell runs directly in the task container ---
tools_file=/opt/agent8088/test-mode-setting/tools.txt
system_file=/opt/agent8088/test-mode-setting/system.md
skills_dir=/opt/agent8088/test-mode-setting/skills
tool_host.execute_shell=1
# view_image here also takes image URLs, as task inputs sometimes are (#21).
image_tool_allow_urls=1
# Defaults only: the Harbor adapter rewrites these per task from its time
# budget (agent8088_harbor/config_edit.py:budget_edits), together with
# max_turn_seconds.
tool_timeout.execute_shell=600
max_tool_timeout_seconds=1800

# --- Workspace and permissions (container is disposable) ---
project_root=/app
shell_cwd=/app
allowed_paths=/
ssrf_allow_private=1
default_permission_mode=full-auto
cron_mode=approve
