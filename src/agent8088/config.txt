# --- plan step auditing ---
# With plan_audit=1, every mutating execute_plan step is verified against the
# real environment by the readonly `auditor` sub-agent before the plan moves on,
# and a failed verification halts the plan. Off by default: it costs one extra
# model call per mutating step, drawn from the same turn budget as the work
# itself. Worth enabling for unattended runs (gateway, cron) where no operator
# is watching, which is exactly where a silently half-done plan does damage.
# plan_audit=0
# With auditing on, a write that FAILS verification is restored to its exact
# pre-step bytes: only verified state persists. Nothing outside the failed step
# is touched, and tools with no undo (shell, docker, cron) say so instead of
# implying a rollback that did not happen.
# plan_audit_revert=1
# Files larger than this are not snapshotted, so they are not reverted — the plan
# output says so rather than implying otherwise.
# plan_audit_revert_max_bytes=1048576
#
# What gets verified: calls that CHANGE something. A tool's permission mode is
# not the test — `shell` is the gate 13 read-only tools go through, so
# git_status, git_log, git_diff and every `ls` used to buy a full auditor run
# each, one of them 111 seconds and 100% of that turn's tokens to confirm that a
# read had read. Reads are skipped now, and an execute_shell is judged on the
# command it was actually given.
#
# Two ceilings, because `/audit on` multiplies a turn's model calls and neither
# max_turn_seconds nor max_turn_tokens is set on a default install:
#
#   plan_audit_max_per_turn     verification calls one top-level turn may spend.
#                               Reaching it says so in the tool result rather
#                               than failing the step — running out of audit
#                               budget is not evidence a write was wrong, so
#                               nothing is reverted. There is no "unlimited"
#                               value: to stop verifying, set plan_audit=0.
#   plan_audit_timeout_seconds  wall clock for ONE auditor sub-run, so a
#                               verification can never outlast the work it
#                               checks.
# plan_audit_max_per_turn=12
# plan_audit_timeout_seconds=120
#
# Every OTHER sub-agent (coder, researcher, test-writer, explore) gets the same
# kind of ceiling, just a wider one — doing the work reasonably takes longer
# than checking it. A sub-agent is handed the parent's REMAINING tokens and
# cost, so delegating still cannot restart the turn's allowance, and its clock
# is capped by whatever the turn itself has left. Before this it inherited
# max_seconds=0 and a test-writer ran `find / -maxdepth 5` until the process
# was killed.
# subagent_timeout_seconds=300

# --- persistent memory ---
# On by default. With memory=1 the agent does two extra things per turn:
#
#   before the turn  the last thing YOU typed is searched against stored
#                    memories, and the top few are added to that turn's system
#                    prompt as context
#   after the answer one extra model call distils durable facts from the
#                    exchange and stores them, deduped against what is known
#
# Retrieval runs two independent searches and fuses them with Reciprocal Rank
# Fusion: SQLite FTS5/BM25 for words (exact flags, filenames, error strings) and
# cosine similarity over embeddings for meaning (finds "uv" from "package
# manager"). Their scores are not comparable, so RRF fuses the ranks instead --
# which is why the vector leg only has to be roughly right, and why a 274MB
# embedder is enough.
#
# The installers pull the embedding model, so semantic recall works from the
# first turn. `/memory` shows what it holds and what the last learning call cost;
# `/memory off` stops both habits without deleting anything, and
# memory_capture=0 below keeps recall while stopping new writes.
#
# Only what YOU type can become a memory or trigger a recall. Tool output -- web
# pages, shell results, file contents -- never does, and a recalled memory can
# never authorise a tool call: check_permission does not read them.
memory=1
#
# Memory engine backend: `native` (default built-in SQLite with BM25 + vector RRF)
# or `mem0` (uses Mem0 library with graph/vector entity management).

# --- web search ---
# One web_search tool, four interchangeable backends, in priority order:
#   searxng  default   self-hosted, no key   (`/search setup` provisions it)
#   ddgs     fallback  keyless, bundled — rotates several engines, no setup
#   tavily   optional  needs TAVILY_API_KEY — takes first priority once set
#   exa      optional  needs EXA_API_KEY — takes priority, behind tavily
# Adding an API key is the signal to prefer that backend, so a configured
# tavily/exa outranks both keyless ones.
#
# web_search_provider decides which one serves:
#
#   auto       (default) At startup, probe and pick the highest-priority
#              backend that can actually serve: a keyed tavily/exa first, else
#              SearXNG if it ANSWERS, else ddgs. The winner is the session's
#              first choice. If it fails mid-session the search falls through to
#              the next backend (a dead SearXNG is then replaced by ddgs, and
#              re-probed every web_search_reprobe_seconds to switch back up).
#              A fallback is announced in the REPL, /status and /doctor.
#   <name>     Pin exactly one of searxng/ddgs/tavily/exa. No auto-selection,
#              no fallback. `/search use <name>` writes this for you.
#   (unset)    Use the whole chain and fall through at call time: a backend
#              that fails (instance down, rate limited) hands off to the next,
#              so web search keeps working — at the cost of a prompt per
#              search, since it may reach a public provider.
#
# Why auto still pins: web_search_no_prompt=1 below only takes effect while a
# LOCAL SearXNG is the pin (see _local_searxng_no_prompt_enabled). With auto,
# SearXNG up means silent local searches; SearXNG down means ddgs serves —
# the bundled keyless backend, which never asks (engine._ddgs_only_chain) — and
# that is true whether SearXNG was down at startup or stopped mid-session. A
# search that skipped approval only ever falls through to ddgs or a local
# SearXNG, never to a keyed vendor or a remote instance.
#
# How often auto re-probes SearXNG while ddgs is standing in (seconds, 0 = never):
# web_search_reprobe_seconds=300
#
# Run `/search status` to see the live chain and `/search doctor` to diagnose.
web_search_provider=auto
#
# SearXNG instance. UNSET by default — no endpoint is assumed for you, so a
# fresh install searches via the keyless ddgs fallback until you choose one.
#
# Three ways to set it:
#   1. `/search` in the REPL — provisions a local container (needs Docker),
#      writes the URL here, and allowlists it. Nothing else to do.
#   2. Point at an instance you already run, on this machine or your LAN:
#        search_base_url=http://127.0.0.1:8888/search?q=
#        search_base_url=http://192.168.1.10:8888/search?q=
#      A LAN host must also be added to ssrf_allow_hosts above.
#   3. Point at a public instance — https:// is REQUIRED (plaintext http:// is
#      accepted only for loopback/private hosts, so queries never cross the
#      internet in the clear):
#        search_base_url=https://searx.example.org/search?q=
#
# The URL must end at `search?q=` with no placeholder — the query is appended.
# The instance must have JSON output enabled (`json` in search.formats), which
# upstream SearXNG disables by default; `/search doctor` reports it if not.
# search_base_url=
#
# Host port for the container `/search` provisions (default 8888). Change it if
# 8888 is already taken. Always published to 127.0.0.1 only — the JSON API is
# unauthenticated, so it is never exposed to the network.
# searxng_host_port=8888
#
# Skip approval only for a pinned loopback or explicitly allowlisted
# private-LAN SearXNG. With an EXPLICIT web_search_provider=searxng it never
# permits a silent fallback: a failed search asks before retrying on public
# ddgs. Under auto, a failed SearXNG falls back to ddgs like a SearXNG that was
# down at startup (see above). Credentials and direct personal identifiers in
# search queries remain blocked.
web_search_no_prompt=1
#
# Results per search (default 5, max 20):
# web_search_results=5
#
# Searches one request may run before it must answer (default 6, 0 = no cap).
# This is a starting allowance: while the latest two searches each keep finding
# mostly new pages, the request earns web_search_extension more searches, up to
# web_search_max_per_turn x web_search_ceiling_multiplier. Rephrasing the same
# question returns the same pages, so it never earns more. A search that
# returns no pages is skipped, neither earning nor blocking more.
# web_search_max_per_turn=6
# web_search_extension=3
# web_search_ceiling_multiplier=3   (1 = fixed cap, no growth)
#
# Optional backends. Add a key and they join the chain automatically; leave them
# unset and they stay out of it entirely. Keys live in the .env store next to
# this file, NEVER here:
#   TAVILY_API_KEY   agent-optimized results with citations  (https://tavily.com)
#   EXA_API_KEY      semantic/neural search                  (https://exa.ai)
#
# The keyless ddgs fallback needs nothing: it ships with agent8088 and is used
# automatically whenever SearXNG is unset, unreachable, or failing. It queries
# several engines in rotation (DuckDuckGo, Brave, Mojeek, Startpage, Yahoo, then
# Wikipedia), retries once with backoff if an engine throttles it, leaves ~2s
# between consecutive searches, and reuses an identical query for 5 minutes — so a
# 202 from one engine no longer ends the search. `/search doctor` shows which
# engines your egress policy currently permits.
#
# Optional outbound proxy for ddgs ONLY (http/https/socks5). DuckDuckGo and some
# other engines block requests from a number of cloud IP ranges, which otherwise
# leaves a VPS install with no keyless search at all. The proxy is treated as any
# other egress destination: it is checked against ssrf_allow_hosts and the egress
# policy, and a proxy the policy forbids is dropped rather than honoured.
# search_proxy=socks5h://127.0.0.1:9150

# --- auto model routing (/model auto) ---
# An opt-in strength ladder, cheapest rung first. Only consulted when the
# selected model is `auto`, `auto:fast` or `auto:smart` -- with any explicit
# model nothing here is read and behaviour is unchanged.
#
# Distinct from fallback_models on purpose: fallback_models answers "what to try
# when the primary is unreachable" (ordered by preference), auto_chain answers
# "what is stronger" (ordered by capability). Escalation moves left to right,
# and only on evidence the turn is failing -- a truncated response, repeated
# invalid tool calls, or no progress. It resets at the start of each turn.
#
# Build one from the providers you hold working keys for with:
#   /model auto setup
# auto_chain=ollama:qwen14b-tooluse-v3,groq:llama-3.3-70b-versatile,anthropic:claude-sonnet-4-6
